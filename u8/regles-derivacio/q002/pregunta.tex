\begin{tria}{regles-tasca}
\itemtria{original}{0,75}{1,25}
Deriva les funcions següents:
\begin{graella}{2}
  \sa f(x)=2x^5-\sqrt[3]{x} & \sa g(x)=3\cdot2^{x}+\log_2 x
\end{graella}

\begin{solucio}
i) $f(x)=2x^5-x^{1/3}$, i per tant $f'(x)=10x^4-\dfrac{1}{3\sqrt[3]{x^2}}$.\\
ii) $g'(x)=3\cdot2^{x}\ln2+\dfrac{1}{x\ln2}$.
\end{solucio}

\itemtria{simplifica-abans}{0,75}{1,25}
Escriu cada funció com una suma de potències de $x$ i deriva-la, sense fer servir la regla del
quocient:
\begin{graella}{2}
  \sa f(x)=\frac{x^3-2\sqrt{x}}{x} & \sa g(x)=\frac{(x+1)^2}{\sqrt{x}}
\end{graella}

\begin{solucio}
i) $f(x)=x^2-2x^{-1/2}$, i per tant $f'(x)=2x+x^{-3/2}=2x+\dfrac{1}{x\sqrt{x}}$.\\
ii) $g(x)=\dfrac{x^2+2x+1}{x^{1/2}}=x^{3/2}+2x^{1/2}+x^{-1/2}$, i per tant
$g'(x)=\dfrac32x^{1/2}+x^{-1/2}-\dfrac12x^{-3/2}=\dfrac32\sqrt{x}+\dfrac{1}{\sqrt{x}}-\dfrac{1}{2x\sqrt{x}}$.
\end{solucio}
\end{tria}

\apartat[1,25]{1}
Deriva les funcions següents:
\begin{graella}{2}
  \sa h(x)=(3x-1)\ln x & \sa k(x)=\frac{e^{x}}{x+1}
\end{graella}

\begin{solucio}
i) $h'(x)=3\ln x+(3x-1)\cdot\dfrac1x=3\ln x+3-\dfrac1x$.\\
ii) $k'(x)=\dfrac{e^{x}(x+1)-e^{x}}{(x+1)^2}=\dfrac{x\,e^{x}}{(x+1)^2}$.
\end{solucio}

\begin{nomesllarg}
\apartat{0,75}
Deriva la funció
\[
m(x)=\frac{\sin x}{x}
\]
i calcula $m'\!\left(\frac{\pi}{2}\right)$.

\begin{solucio}
$m'(x)=\dfrac{x\cos x-\sin x}{x^2}$, i
$m'\!\left(\dfrac{\pi}{2}\right)=\dfrac{\frac{\pi}{2}\cdot0-1}{\pi^2/4}=-\dfrac{4}{\pi^2}$.
\end{solucio}
\end{nomesllarg}
