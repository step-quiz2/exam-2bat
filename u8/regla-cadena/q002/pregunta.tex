\apartat[1,25]{0,75}
Deriva les funcions següents:
\begin{graella}{2}
  \sa f(x)=\left(2x^3+5\right)^5 & \sa g(x)=e^{3x^2-x}
\end{graella}

\begin{solucio}
i) $f'(x)=5\left(2x^3+5\right)^4\cdot6x^2=30x^2\left(2x^3+5\right)^4$.\\
ii) $g'(x)=(6x-1)\,e^{3x^2-x}$.
\end{solucio}

\apartat[1,25]{1}
Deriva les funcions següents i simplifica'n el resultat:
\begin{graella}{2}
  \sa h(x)=\ln\left(5x^2+2\right) & \sa k(x)=\sqrt{x^2+9}
\end{graella}

\begin{solucio}
i) $h'(x)=\dfrac{10x}{5x^2+2}$.\\
ii) $k(x)=\left(x^2+9\right)^{1/2}$, i per tant
$k'(x)=\dfrac12\left(x^2+9\right)^{-1/2}\cdot2x=\dfrac{x}{\sqrt{x^2+9}}$.
\end{solucio}

\begin{nomesllarg}
\apartat{0,75}
Deriva la funció
\[
m(x)=\cos^3(2x)
\]
i calcula $m'\!\left(\frac{\pi}{6}\right)$.

\begin{solucio}
$m'(x)=3\cos^2(2x)\cdot\left(-\sin(2x)\right)\cdot2=-6\cos^2(2x)\sin(2x)$.\\
En $x=\frac{\pi}{6}$, $2x=\frac{\pi}{3}$:
$m'\!\left(\dfrac{\pi}{6}\right)=-6\cdot\left(\dfrac12\right)^2\cdot\dfrac{\sqrt3}{2}
=-\dfrac{3\sqrt3}{4}$.
\end{solucio}
\end{nomesllarg}
