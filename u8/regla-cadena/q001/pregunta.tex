\begin{tria}{cadena-tasca}
\itemtria{original}{0,75}{1,25}
Deriva les funcions següents:
\begin{graella}{2}
  \sa f(x)=\left(3x^2-1\right)^4 & \sa g(x)=e^{x^2+2x}
\end{graella}

\begin{solucio}
i) $f'(x)=4\left(3x^2-1\right)^3\cdot6x=24x\left(3x^2-1\right)^3$.\\
ii) $g'(x)=(2x+2)\,e^{x^2+2x}$.
\end{solucio}

\itemtria{composicio-amb-valors}{0,75}{1,25}
De dues funcions derivables $u$ i $v$ se sap que
\[
u(1)=2,\qquad u'(1)=3,\qquad v(2)=-1,\qquad v'(2)=4 .
\]
Calcula la derivada de $v\bigl(u(x)\bigr)$ i la de $\bigl(u(x)\bigr)^3$ en $x=1$.

\begin{solucio}
Per la regla de la cadena, $\bigl[v\bigl(u(x)\bigr)\bigr]'=v'\bigl(u(x)\bigr)\cdot u'(x)$. En $x=1$:
$v'\bigl(u(1)\bigr)\cdot u'(1)=v'(2)\cdot3=4\cdot3=12$.\\
I $\bigl[\bigl(u(x)\bigr)^3\bigr]'=3\bigl(u(x)\bigr)^2\cdot u'(x)$, que en $x=1$ val $3\cdot2^2\cdot3=36$.\\
Compte: el valor $v(2)=-1$ no hi intervé; el que compta és la derivada de $v$ en el punt $u(1)=2$.
\end{solucio}
\end{tria}

\apartat[1,25]{1}
Deriva les funcions següents i simplifica'n el resultat:
\begin{graella}{2}
  \sa h(x)=\ln\left(x^3+2x\right) & \sa k(x)=\ln\sqrt{2x^2+1}
\end{graella}

\begin{solucio}
i) $h'(x)=\dfrac{3x^2+2}{x^3+2x}$.\\
ii) Com que $k(x)=\tfrac12\ln\left(2x^2+1\right)$,
$k'(x)=\dfrac12\cdot\dfrac{4x}{2x^2+1}=\dfrac{2x}{2x^2+1}$.
\end{solucio}

\begin{nomesllarg}
\apartat{0,75}
Deriva la funció
\[
m(x)=\sin^2(3x)
\]
i calcula $m'\!\left(\frac{\pi}{12}\right)$.

\begin{solucio}
$m'(x)=2\sin(3x)\cdot\cos(3x)\cdot3=6\sin(3x)\cos(3x)$.\\
En $x=\frac{\pi}{12}$, $3x=\frac{\pi}{4}$:
$m'\!\left(\dfrac{\pi}{12}\right)=6\cdot\dfrac{\sqrt2}{2}\cdot\dfrac{\sqrt2}{2}=3$.
\end{solucio}
\end{nomesllarg}
