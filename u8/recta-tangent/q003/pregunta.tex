\begin{apartats}

\apartat[1,25]{1}
Troba l'equació de la recta tangent a la gràfica de
\[
f(x)=e^{2x-2}
\]
en el punt d'abscissa $x=1$.

\begin{solucio}
$f(1)=e^{0}=1$ i $f'(x)=2e^{2x-2}$, d'on $f'(1)=2$.\\
Tangent: $y-1=2(x-1)$, és a dir $y=2x-1$.
\end{solucio}

\begin{nomesllarg}
\apartat{0,75}
Troba l'equació de la recta normal a la gràfica de $f$ en el mateix punt.

\begin{solucio}
El pendent de la normal és $-\dfrac{1}{f'(1)}=-\dfrac12$.\\
Normal: $y-1=-\dfrac12(x-1)$, és a dir $y=-\dfrac{x}{2}+\dfrac32$.
\end{solucio}
\end{nomesllarg}

\apartat[1,25]{0,75}
La recta tangent a la gràfica de $g(x)=ax^2+3$ en el punt d'abscissa $x=2$ té pendent $8$.
Troba $a$ i l'equació d'aquesta tangent.

\begin{solucio}
$g'(x)=2ax$ i $g'(2)=4a=8$, d'on $\boxed{a=2}$.\\
Aleshores $g(2)=2\cdot4+3=11$, i la tangent és $y-11=8(x-2)$, és a dir $y=8x-5$.
\end{solucio}

\end{apartats}
