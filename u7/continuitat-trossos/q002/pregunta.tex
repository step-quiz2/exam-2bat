\begin{tria}{trosos-tasca-a}
\itemtria{original}{1,25}{2,5}
Estudia la continuïtat de la funció següent i classifica'n les discontinuïtats.
\[
f(x)=\begin{cases}
  x+2 & \si{x<-1},\\[3pt]
  x^2 & \si{-1\le x\le 2},\\[3pt]
  3+\ln(x-1) & \si{x>2}.
\end{cases}
\]

\begin{solucio}
Les dues primeres branques són polinòmiques. La tercera és contínua per a $x>1$, i per
tant a $(2,+\infty)$. Cal estudiar els enganxaments.\\
$x=-1$: $\lim_{x\to-1^-}f(x)=1$, $f(-1)=1$ i $\lim_{x\to-1^+}f(x)=1$. \textbf{És contínua.}\\
$x=2$: $f(2)=4=\lim_{x\to2^-}f(x)$, però $\lim_{x\to2^+}f(x)=3+\ln1=3$.
Laterals finits i diferents: \textbf{salt finit} (de salt $1$).\\
Per tant, $f$ és contínua a $\mathbb{R}\setminus\{2\}$.
\end{solucio}

\itemtria{branca-interna}{1,25}{2,5}
Estudia la continuïtat de la funció següent i classifica'n les discontinuïtats.
\[
g(x)=\begin{cases}
  x+2 & \si{x<-1},\\[3pt]
  x^2 & \si{-1\le x\le 2},\\[6pt]
  \dfrac{4}{x-3} & \si{x>2}.
\end{cases}
\]

\begin{solucio}
Les dues primeres branques són polinòmiques. La tercera és contínua a tot el seu interval
excepte on s'anul·la el denominador, $x=3$, que és \textbf{dins} de $(2,+\infty)$: cal
estudiar els dos enganxaments i també aquest punt.\\
$x=-1$: $\lim_{x\to-1^-}g(x)=1$, $g(-1)=1$ i $\lim_{x\to-1^+}g(x)=1$. \textbf{És contínua.}\\
$x=2$: $g(2)=4=\lim_{x\to2^-}g(x)$, però $\lim_{x\to2^+}g(x)=\dfrac{4}{2-3}=-4$.
Laterals finits i diferents: \textbf{salt finit} (de salt $8$).\\
$x=3$: $g(3)$ no existeix, i els laterals valen $-\infty$ (per l'esquerra) i $+\infty$ (per la
dreta): \textbf{salt infinit} (asímptota vertical $x=3$).\\
Per tant, $g$ és contínua a $\mathbb{R}\setminus\{2,3\}$.
\end{solucio}
\end{tria}

\begin{nomesllarg}
\apartat{1,25}
Escriu la funció
\[
h(x)=\frac{x^2-1}{|x-1|}
\]
com una funció definida a trossos, sense valor absolut. Estudia'n la continuïtat i
classifica les discontinuïtats que presenti.

\begin{solucio}
$h$ no està definida en $x=1$.\\
Si $x>1$: $|x-1|=x-1$ i $h(x)=\dfrac{(x-1)(x+1)}{x-1}=x+1$.\\
Si $x<1$: $|x-1|=-(x-1)$ i $h(x)=\dfrac{(x-1)(x+1)}{-(x-1)}=-x-1$.
\[
h(x)=\begin{cases} -x-1 & \si{x<1},\\ x+1 & \si{x>1}.\end{cases}
\]
Cada branca és polinòmica, i per tant $h$ és contínua a $\mathbb{R}\setminus\{1\}$.
En $x=1$: $\lim_{x\to1^-}h(x)=-2$ i $\lim_{x\to1^+}h(x)=2$. Laterals finits i
diferents: \textbf{salt finit}.
\end{solucio}
\end{nomesllarg}

\end{apartats}
