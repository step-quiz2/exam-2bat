\begin{apartats}

\apartat[1,25]{0,75}
Calcula el límit següent:
\[
\lim_{x\to-2}\frac{x^2+x-2}{x^2+3x+2}
\]

\begin{solucio}
És una indeterminació $0/0$. Factoritzem i simplifiquem:
\[
\frac{(x+2)(x-1)}{(x+1)(x+2)}=\frac{x-1}{x+1}\;\xrightarrow[x\to-2]{}\;\frac{-3}{-1}=3.
\]
\end{solucio}

\begin{nomesllarg}
\apartat{0,75}
Estudia si existeix el límit següent. Si no existeix, justifica-ho amb els límits laterals.
\[
\lim_{x\to1}\frac{x^2-1}{x^2-2x+1}
\]

\begin{solucio}
És una indeterminació $0/0$: $\dfrac{(x-1)(x+1)}{(x-1)^2}=\dfrac{x+1}{x-1}$, que tendeix a
$\dfrac{2}{0}$. Els límits laterals valen $-\infty$ (per l'esquerra) i $+\infty$ (per la
dreta), i per tant el límit \textbf{no existeix}.
\end{solucio}
\end{nomesllarg}

\apartat[1,25]{1}
Donada la funció
\[
g(x)=\begin{cases} x^2+1 & \si{x<2},\\[4pt] \dfrac{12}{x+1} & \si{x\ge 2},\end{cases}
\]
calcula, si existeixen, els límits següents:
\begin{graella}{2}
  \sa \lim_{x\to-1}g(x) & \sa \lim_{x\to2}g(x)
\end{graella}

\begin{solucio}
i) Com que $-1<2$, hi actua la branca $x^2+1$: el límit val $2$. La branca
$\dfrac{12}{x+1}$ no hi intervé, tot i no estar definida en $x=-1$.\\
ii) Per l'esquerra, $x^2+1\to5$; per la dreta, $\dfrac{12}{x+1}\to4$. Els laterals són
diferents, i per tant el límit \textbf{no existeix}.
\end{solucio}

\end{apartats}
